// ==================== 天气配置 ====================
// 添加到 CONFIG 对象中：
// CACHE_DURATION: 2 * 60 * 60 * 1000, // 天气缓存 2 小时
// WEATHER_TIMEOUT: 10000,
// GEO_TIMEOUT: 6000,

// ==================== 天气数据源 ====================
const WEATHER_SOURCES = {
    openMeteo: {
        name: 'Open-Meteo',
        async fetch(coords) {
            const url = `https://api.open-meteo.com/v1/forecast?latitude=${coords.lat}&longitude=${coords.lon}¤t=temperature_2m,weather_code,wind_speed_10m&daily=temperature_2m_max,temperature_2m_min&timezone=auto`;
            const res = await fetchWithTimeout(url, {}, 8000);
            if (!res.ok) throw new Error(`HTTP ${res.status}`);
            const data = await res.json();
            if (!data.current || typeof data.current.temperature_2m !== 'number') throw new Error('数据异常');
            return {
                temp: Math.round(data.current.temperature_2m),
                desc: getWeatherDesc(data.current.weather_code),
                wind: Math.round(data.current.wind_speed_10m),
                high: Math.round(data.daily.temperature_2m_max[0]),
                low: Math.round(data.daily.temperature_2m_min[0])
            };
        }
    },
    wttrIn: {
        name: 'wttr.in',
        async fetch() {
            const url = 'https://wttr.in/?format=j1&lang=zh';
            const res = await fetchWithTimeout(url, {}, CONFIG.WEATHER_TIMEOUT);
            if (!res.ok) throw new Error(`HTTP ${res.status}`);
            const data = await res.json();
            const current = data.current_condition?.[0];
            const today = data.weather?.[0];
            if (!current) throw new Error('无天气数据');
            return {
                temp: parseInt(current.temp_C),
                desc: current.lang_zh?.[0]?.value || current.weatherDesc?.[0]?.value || '未知',
                wind: parseInt(current.windspeedKmph) || 0,
                high: today ? parseInt(today.maxtempC) : null,
                low: today ? parseInt(today.mintempC) : null
            };
        }
    }
};

// ==================== 天气工具函数 ====================
function getWeatherDesc(code) {
    const map = {
        0: '晴朗', 1: '晴间多云', 2: '多云', 3: '阴天',
        45: '雾', 48: '雾凇', 51: '毛毛雨', 53: '中毛毛雨', 55: '大毛毛雨',
        61: '小雨', 63: '中雨', 65: '大雨', 71: '小雪', 73: '中雪', 75: '大雪',
        80: '阵雨', 81: '强阵雨', 82: '暴雨', 95: '雷雨', 96: '雷伴冰雹', 99: '强雷暴'
    };
    return map[code] ?? '多云';
}

// ==================== 天气缓存 ====================
function loadWeatherCache() {
    const cache = storageGet('weatherCache');
    if (!cache) return null;
    return (Date.now() - cache.timestamp) < CONFIG.CACHE_DURATION ? cache : null;
}
function saveWeatherCache(data, name) {
    storageSet('weatherCache', { data, name, timestamp: Date.now() });
}

// ==================== 地理定位 ====================
function getCurrentPosition() {
    return new Promise((resolve, reject) => {
        if (!navigator.geolocation) { reject(new Error('不支持定位')); return; }
        if (window.location.protocol === 'file:') { reject(new Error('本地文件不支持定位')); return; }
        navigator.geolocation.getCurrentPosition(
            pos => resolve({ lat: pos.coords.latitude, lon: pos.coords.longitude }),
            reject,
            { timeout: CONFIG.GEO_TIMEOUT, enableHighAccuracy: false, maximumAge: 300000 }
        );
    });
}

// ==================== 天气渲染 ====================
function renderWeather(data, name) {
    dom.location.textContent = name;
    dom.temperature.textContent = `${data.temp}°`;
    dom.highLow.textContent = data.high !== null ? `H: ${data.high}° L: ${data.low}°` : '--';
    dom.wind.textContent = `${data.wind} km/h`;
    dom.weatherDesc.textContent = data.desc;
}

// ==================== 天气主流程 ====================
async function initWeather() {
    if (state.isWeatherLoading) return;
    state.isWeatherLoading = true;

    const cache = loadWeatherCache();
    if (cache) { renderWeather(cache.data, cache.name); }

    let weatherData = null;
    const locationName = '本地';

    try {
        try {
            const coords = await getCurrentPosition();
            weatherData = await WEATHER_SOURCES.openMeteo.fetch(coords);
        } catch (e) {
            console.warn('Open-Meteo 链路失败:', e.message);
        }

        if (!weatherData) {
            try {
                weatherData = await WEATHER_SOURCES.wttrIn.fetch();
            } catch (e) {
                console.warn('wttr.in 失败:', e.message);
            }
        }

        if (weatherData) {
            renderWeather(weatherData, locationName);
            saveWeatherCache(weatherData, locationName);
            dom.weatherError.classList.add('hidden');
        } else {
            throw new Error('所有天气源均不可用');
        }
    } catch (e) {
        console.error('天气加载失败:', e);
        if (!cache) {
            dom.location.textContent = '天气暂不可用';
            dom.temperature.textContent = '--°';
            dom.highLow.textContent = '--'; dom.wind.textContent = '--'; dom.weatherDesc.textContent = '--';
        } else {
            dom.weatherError.textContent = '已离线 · 显示缓存数据';
            dom.weatherError.classList.remove('hidden');
        }
    } finally {
        state.isWeatherLoading = false;
    }
}
